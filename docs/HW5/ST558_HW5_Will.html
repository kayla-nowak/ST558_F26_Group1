  url <- paste0("https://newsapi.org/v2/everything?q=", URLencode(query),
  "&from=", from_date,
  "&language=en&pageSize=50&apiKey=", api_key)
  
  result <- GET(url)
  
  parsed_result <- fromJSON(content(result, as = "text", encoding = "UTF-8"))
  
  parsed_result$articles
}
