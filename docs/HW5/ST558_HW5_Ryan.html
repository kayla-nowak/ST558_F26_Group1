  1. Now write a quick function that allows the user to easily query this API. The inputs to the function should be the title/subject to search for, a time period to search from, and an API key.
query_function <- function(subject, date, api_key) {
  response <- GET(
    url = "https://newsapi.org/v2/everything",
    query = list(
      q = subject,
      from = date,
      language = "en",
      pageSize = 50,
      apiKey = api_key
    )
  )
  
  parsed <- fromJSON(rawToChar(response$content))
  as_tibble(parsed$articles)
}
